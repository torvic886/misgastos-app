package com.misgastos.controller;

import javax.crypto.spec.PBEKeySpec;

/**
 * Clave propia de la pantalla "Respaldo de datos". No depende de ningún usuario del
 * programa: solo quien la sabe puede entrar, aunque sea administrador.
 *
 * La clave NO se guarda en texto: se guarda un resumen (PBKDF2 con sal) en
 * MisGastos\respaldo.clave. La primera vez que se abre la pantalla se pide crearla.
 */
public final class ClaveRespaldo {

    private static final int ITERACIONES = 120_000;
    private static final int BITS = 256;
    private static final int LARGO_SAL = 16;
    private static final int LARGO_MINIMO = 4;

    private ClaveRespaldo() {
    }

    private static Path archivo() {
        return Paths.get(System.getProperty("user.home"), "MisGastos", "respaldo.clave");
    }

    /** Devuelve true si la persona puede entrar (clave correcta, o clave recién creada). */
    public static boolean pedirAcceso(Window ventana) {
        if (!Files.exists(archivo())) {
            return crearClave(ventana);
        }
        return verificarClave(ventana);
    }

    // ==================== CREAR ====================

    private static boolean crearClave(Window ventana) {
        String aviso = "Aún no hay una clave para el respaldo. Crea una que solo tú conozcas.\n"
                + "Mínimo " + LARGO_MINIMO + " caracteres.";
        String error = null;
        while (true) {
            Optional<String[]> r = dialogo(ventana, "Crear clave de respaldo", aviso, true, error);
            if (r.isEmpty()) {
                return false;
            }
            String uno = r.get()[0];
            String dos = r.get()[1];
            if (uno.length() < LARGO_MINIMO) {
                error = "La clave debe tener al menos " + LARGO_MINIMO + " caracteres.";
            } else if (!uno.equals(dos)) {
                error = "Las dos claves no coinciden.";
            } else {
                try {
                    guardar(uno.toCharArray());
                    return true;
                } catch (Exception e) {
                    alerta(ventana, Alert.AlertType.ERROR, "No se pudo guardar la clave", e.getMessage());
                    return false;
                }
            }
        }
    }

    private static void guardar(char[] clave) throws Exception {
        byte[] sal = new byte[LARGO_SAL];
        new SecureRandom().nextBytes(sal);
        byte[] resumen = resumir(clave, sal);
        String texto = "v1:" + Base64.getEncoder().encodeToString(sal)
                + ":" + Base64.getEncoder().encodeToString(resumen);
        Files.createDirectories(archivo().getParent());
        Files.writeString(archivo(), texto, StandardCharsets.UTF_8);
    }

    // ==================== VERIFICAR ====================

    private static boolean verificarClave(Window ventana) {
        byte[] sal;
        byte[] esperado;
        try {
            String[] partes = Files.readString(archivo(), StandardCharsets.UTF_8).trim().split(":");
            if (partes.length != 3 || !"v1".equals(partes[0])) {
                throw new IllegalStateException("formato no reconocido");
            }
            sal = Base64.getDecoder().decode(partes[1]);
            esperado = Base64.getDecoder().decode(partes[2]);
        } catch (Exception e) {
            alerta(ventana, Alert.AlertType.ERROR, "Archivo de clave dañado",
                    "No se puede leer la clave guardada (" + archivo() + ").");
            return false;
        }

        String error = null;
        while (true) {
            Optional<String[]> r = dialogo(ventana, "Respaldo de datos",
                    "Escribe la clave de respaldo para continuar.", false, error);
            if (r.isEmpty()) {
                return false;
            }
            try {
                byte[] intento = resumir(r.get()[0].toCharArray(), sal);
                if (MessageDigest.isEqual(intento, esperado)) {
                    return true;
                }
            } catch (Exception e) {
                alerta(ventana, Alert.AlertType.ERROR, "Error", e.getMessage());
                return false;
            }
            error = "Clave incorrecta.";
        }
    }

    private static byte[] resumir(char[] clave, byte[] sal) throws Exception {
        PBEKeySpec spec = new PBEKeySpec(clave, sal, ITERACIONES, BITS);
        try {
            return SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256").generateSecret(spec).getEncoded();
        } finally {
            spec.clearPassword();
        }
    }

    // ==================== VENTANAS ====================

    private static Optional<String[]> dialogo(Window ventana, String titulo, String texto,
                                              boolean repetir, String error) {
        Dialog<String[]> d = new Dialog<>();
        if (ventana != null) {
            d.initOwner(ventana);
        }
        d.setTitle(titulo);
        d.setHeaderText(texto);

        PasswordField uno = new PasswordField();
        uno.setPromptText("Clave");
        PasswordField dos = new PasswordField();
        dos.setPromptText("Repite la clave");

        VBox caja = new VBox(10);
        caja.setPadding(new Insets(10, 0, 0, 0));
        caja.getChildren().add(uno);
        if (repetir) {
            caja.getChildren().add(dos);
        }
        if (error != null) {
            Label l = new Label(error);
            l.setStyle("-fx-text-fill: #dc2626; -fx-font-weight: bold;");
            caja.getChildren().add(l);
        }

        ButtonType aceptar = new ButtonType("Aceptar", ButtonBar.ButtonData.OK_DONE);
        ButtonType cancelar = new ButtonType("Cancelar", ButtonBar.ButtonData.CANCEL_CLOSE);
        d.getDialogPane().getButtonTypes().setAll(aceptar, cancelar);
        d.getDialogPane().setContent(caja);
        estilos(d.getDialogPane());
        d.setResultConverter(b -> b == aceptar ? new String[]{uno.getText(), dos.getText()} : null);

        Platform.runLater(uno::requestFocus);
        return d.showAndWait();
    }

    private static void alerta(Window ventana, Alert.AlertType tipo, String titulo, String mensaje) {
        Alert a = new Alert(tipo);
        if (ventana != null) {
            a.initOwner(ventana);
        }
        a.setTitle(titulo);
        a.setHeaderText(null);
        a.setContentText(mensaje);
        estilos(a.getDialogPane());
        a.showAndWait();
    }

    private static void estilos(javafx.scene.control.DialogPane panel) {
        try {
            panel.getStylesheets().add(ClaveRespaldo.class.getResource("/css/styles.css").toExternalForm());
        } catch (Exception ignorada) {
            // sin estilos, funciona igual
        }
    }
}

package com.misgastos.controller;

/**
 * Pantalla "Respaldo de datos": abrir la carpeta de datos, exportar una copia
 * de la base y importar una copia.
 *
 * Importar NO reemplaza la base mientras el programa está abierto (Windows no deja
 * reemplazar un archivo en uso). La copia se deja preparada y MisGastosLauncher la
 * aplica en el siguiente arranque, después de guardar un respaldo de lo actual.
 */
@Slf4j
@Component
public class RespaldoController {

    private static final Locale CO = Locale.forLanguageTag("es-CO");
    private static final DateTimeFormatter FECHA_ARCHIVO = DateTimeFormatter.ofPattern("yyyy-MM-dd_HH-mm-ss");
    private static final DateTimeFormatter FECHA_PANTALLA = DateTimeFormatter.ofPattern("dd/MM/yyyy HH:mm");

    @FXML private Label lblCarpeta;
    @FXML private Label lblBase;
    @FXML private Label lblRespaldos;
    @FXML private Label lblPendiente;
    @FXML private Button btnCancelarPendiente;

    @Autowired
    private DataSource dataSource;

    // ==================== RUTAS ====================

    private Path carpetaDatos() {
        return Paths.get(System.getProperty("user.home"), "MisGastos");
    }

    private Path archivoBase() {
        return carpetaDatos().resolve("misgastos.db");
    }

    private Path carpetaRespaldos() {
        return carpetaDatos().resolve("respaldos");
    }

    private Path archivoPendiente() {
        return carpetaDatos().resolve(MisGastosLauncher.ARCHIVO_IMPORTACION);
    }

    // ==================== PANTALLA ====================

    @FXML
    public void initialize() {
        actualizarResumen();
    }

    private void actualizarResumen() {
        lblCarpeta.setText(carpetaDatos().toString());

        try {
            double mb = Files.size(archivoBase()) / 1048576.0;
            lblBase.setText(String.format(CO, "misgastos.db  (%.2f MB)", mb));
        } catch (Exception e) {
            lblBase.setText("No se encontró la base de datos");
        }

        List<Path> respaldos = listarRespaldos();
        if (respaldos.isEmpty()) {
            lblRespaldos.setText("Todavía no hay copias automáticas.");
        } else {
            Path ultimo = respaldos.get(respaldos.size() - 1);
            String cuando;
            try {
                cuando = LocalDateTime
                        .ofInstant(Instant.ofEpochMilli(Files.getLastModifiedTime(ultimo).toMillis()), ZoneId.systemDefault())
                        .format(FECHA_PANTALLA);
            } catch (Exception e) {
                cuando = ultimo.getFileName().toString();
            }
            lblRespaldos.setText(respaldos.size() + " copia(s) automática(s). La última: " + cuando);
        }

        boolean hayPendiente = Files.exists(archivoPendiente());
        lblPendiente.setVisible(hayPendiente);
        lblPendiente.setManaged(hayPendiente);
        btnCancelarPendiente.setVisible(hayPendiente);
        btnCancelarPendiente.setManaged(hayPendiente);
    }

    private List<Path> listarRespaldos() {
        Path carpeta = carpetaRespaldos();
        if (!Files.isDirectory(carpeta)) {
            return List.of();
        }
        try (Stream<Path> archivos = Files.list(carpeta)) {
            return archivos
                    .filter(p -> {
                        String n = p.getFileName().toString();
                        return n.startsWith("misgastos_") && n.endsWith(".db");
                    })
                    .sorted()
                    .toList();
        } catch (Exception e) {
            return List.of();
        }
    }

    // ==================== ABRIR CARPETAS ====================

    @FXML
    public void handleAbrirCarpetaDatos() {
        abrirCarpeta(carpetaDatos());
    }

    @FXML
    public void handleAbrirCarpetaRespaldos() {
        abrirCarpeta(carpetaRespaldos());
    }

    private void abrirCarpeta(Path carpeta) {
        try {
            Files.createDirectories(carpeta);
            String os = System.getProperty("os.name", "").toLowerCase();
            ProcessBuilder pb;
            if (os.contains("win")) {
                pb = new ProcessBuilder("explorer.exe", carpeta.toString());
            } else if (os.contains("mac")) {
                pb = new ProcessBuilder("open", carpeta.toString());
            } else {
                pb = new ProcessBuilder("xdg-open", carpeta.toString());
            }
            pb.start();
        } catch (Exception e) {
            alerta(Alert.AlertType.ERROR, "No se pudo abrir la carpeta",
                    "Puedes abrirla a mano desde el Explorador de archivos:\n" + carpeta);
        }
    }

    // ==================== EXPORTAR ====================

    @FXML
    public void handleExportar() {
        FileChooser selector = new FileChooser();
        selector.setTitle("Exportar copia de MisGastos");
        selector.setInitialFileName("misgastos_" + LocalDateTime.now().format(FECHA_ARCHIVO) + ".db");
        selector.getExtensionFilters().add(
                new FileChooser.ExtensionFilter("Base de datos MisGastos (*.db)", "*.db"));

        File elegido = selector.showSaveDialog(ventana());
        if (elegido == null) {
            return;
        }

        Path destino = elegido.toPath();
        if (!destino.getFileName().toString().toLowerCase().endsWith(".db")) {
            destino = destino.resolveSibling(destino.getFileName() + ".db");
        }

        try {
            if (Files.isSameFile(archivoBase(), destino)) {
                alerta(Alert.AlertType.WARNING, "Elige otro lugar",
                        "No puedes exportar encima de la base que el programa está usando.");
                return;
            }
        } catch (Exception ignorada) {
            // el destino aún no existe: es lo normal
        }

        try {
            Files.deleteIfExists(destino);
            copiarBaseConsistente(destino);
            alerta(Alert.AlertType.INFORMATION, "Copia exportada",
                    "La copia quedó guardada en:\n" + destino
                            + "\n\nPuedes llevarla a otro computador o guardarla como respaldo.");
        } catch (Exception e) {
        	log.error("No se pudo exportar la copia de la base de datos", e);
            alerta(Alert.AlertType.ERROR, "No se pudo exportar", "Motivo: " + e.getMessage());
        }
    }

    /**
     * Copia la base mientras el programa está abierto. VACUUM INTO genera una copia
     * completa y consistente aunque haya datos recién guardados. Si no estuviera
     * disponible, se copia el archivo directamente.
     */
    private void copiarBaseConsistente(Path destino) throws Exception {
        try (Connection c = dataSource.getConnection(); Statement st = c.createStatement()) {
            st.execute("VACUUM INTO '" + destino.toString().replace("'", "''") + "'");
        } catch (Exception e) {
        	log.warn("VACUUM INTO no disponible, se copia el archivo: {}", e.getMessage());
            Files.copy(archivoBase(), destino, StandardCopyOption.REPLACE_EXISTING);
        }
    }

    // ==================== IMPORTAR ====================

    @FXML
    public void handleImportar() {
        FileChooser selector = new FileChooser();
        selector.setTitle("Elige la copia que quieres importar");
        selector.getExtensionFilters().add(
                new FileChooser.ExtensionFilter("Base de datos MisGastos (*.db)", "*.db"));
        if (Files.isDirectory(carpetaRespaldos())) {
            selector.setInitialDirectory(carpetaRespaldos().toFile());
        }

        File elegido = selector.showOpenDialog(ventana());
        if (elegido == null) {
            return;
        }
        Path origen = elegido.toPath();

        try {
            if (Files.isSameFile(origen, archivoBase())) {
                alerta(Alert.AlertType.WARNING, "Ese es el archivo actual",
                        "Elige una copia guardada en otro lugar (por ejemplo, una de la carpeta de respaldos).");
                return;
            }
        } catch (Exception ignorada) {
            // seguimos
        }

        Path preparado = archivoPendiente();
        try {
            Files.copy(origen, preparado, StandardCopyOption.REPLACE_EXISTING);

            String problema = validarBase(preparado);
            if (problema != null) {
                Files.deleteIfExists(preparado);
                alerta(Alert.AlertType.ERROR, "Esa copia no sirve", problema);
                actualizarResumen();
                return;
            }

            Alert confirmar = new Alert(Alert.AlertType.CONFIRMATION);
            confirmar.setTitle("Importar copia");
            confirmar.setHeaderText("Se reemplazarán TODOS los datos actuales");
            confirmar.setContentText(
                    "Los datos de este archivo sustituirán a los que tienes ahora.\n\n"
                  + "• El cambio se aplica la próxima vez que abras el programa.\n"
                  + "• Antes de aplicarlo se guarda una copia de lo actual en la carpeta de respaldos.\n"
                  + "• Si registras gastos antes de cerrar, quedarán en esa copia, no en los datos nuevos.\n\n"
                  + "¿Quieres continuar?");
            aplicarEstilos(confirmar);
            ButtonType si = new ButtonType("Sí, importar", javafx.scene.control.ButtonBar.ButtonData.OK_DONE);
            ButtonType no = new ButtonType("Cancelar", javafx.scene.control.ButtonBar.ButtonData.CANCEL_CLOSE);
            confirmar.getButtonTypes().setAll(si, no);

            Optional<ButtonType> respuesta = confirmar.showAndWait();
            if (respuesta.isEmpty() || respuesta.get() != si) {
                Files.deleteIfExists(preparado);
                actualizarResumen();
                return;
            }

            actualizarResumen();
            alerta(Alert.AlertType.INFORMATION, "Copia lista para aplicar",
                    "Cierra el programa y vuelve a abrirlo. Al iniciar se cargarán los datos de la copia importada.");
        } catch (Exception e) {
        	log.error("No se pudo importar la copia de la base de datos", e);
            try {
                Files.deleteIfExists(preparado);
            } catch (Exception ignorada) {
                // nada más que hacer
            }
            alerta(Alert.AlertType.ERROR, "No se pudo importar", "Motivo: " + e.getMessage());
            actualizarResumen();
        }
    }

    @FXML
    public void handleCancelarPendiente() {
        try {
            Files.deleteIfExists(archivoPendiente());
        } catch (Exception e) {
            alerta(Alert.AlertType.ERROR, "No se pudo cancelar", "Motivo: " + e.getMessage());
        }
        actualizarResumen();
    }

    /**
     * Revisa que el archivo sea una base de MisGastos sana.
     * Devuelve null si está bien, o el motivo en palabras sencillas si no.
     */
    private String validarBase(Path archivo) {
        try (InputStream in = Files.newInputStream(archivo)) {
            byte[] cabecera = in.readNBytes(16);
            byte[] esperada = "SQLite format 3\0".getBytes(java.nio.charset.StandardCharsets.ISO_8859_1);
            if (!Arrays.equals(cabecera, esperada)) {
                return "El archivo no es una base de datos de MisGastos.";
            }
        } catch (Exception e) {
            return "No se pudo leer el archivo: " + e.getMessage();
        }

        String url = "jdbc:sqlite:" + archivo.toString().replace('\\', '/');
        try (Connection c = DriverManager.getConnection(url); Statement st = c.createStatement()) {
            try (ResultSet rs = st.executeQuery("PRAGMA quick_check")) {
                if (!rs.next() || !"ok".equalsIgnoreCase(rs.getString(1))) {
                    return "El archivo está dañado, no se puede importar.";
                }
            }
            try (ResultSet rs = st.executeQuery(
                    "SELECT COUNT(*) FROM sqlite_master WHERE type = 'table' AND name = 'usuarios'")) {
                if (!rs.next() || rs.getInt(1) == 0) {
                    return "El archivo es una base de datos, pero no es de MisGastos.";
                }
            }
        } catch (Exception e) {
            return "No se pudo revisar el archivo: " + e.getMessage();
        }
        return null;
    }

    private Window ventana() {
        return lblCarpeta.getScene() == null ? null : lblCarpeta.getScene().getWindow();
    }

    private void alerta(Alert.AlertType tipo, String titulo, String mensaje) {
        Alert a = new Alert(tipo);
        a.setTitle(titulo);
        a.setHeaderText(null);
        a.setContentText(mensaje);
        aplicarEstilos(a);
        a.showAndWait();
    }

    private void aplicarEstilos(Alert a) {
        try {
            a.getDialogPane().getStylesheets().add(
                    getClass().getResource("/css/styles.css").toExternalForm());
        } catch (Exception ignorada) {
            // sin estilos, el diálogo funciona igual
        }
    }
}

package com.misgastos;

import javafx.application.Application;

import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.List;
import java.util.stream.Stream;

/**
 * Punto de entrada real de la aplicación.
 * No extiende Application (necesario para ejecutar desde el fat jar y el instalador).
 * Prepara la carpeta y la base de datos ANTES de que Spring intente conectarse.
 */
public class MisGastosLauncher {

    private static final String PLANTILLA = "/plantilla/misgastos.db";

    // Respaldo automático: carpeta dentro de MisGastos y cuántas copias se conservan
    private static final String CARPETA_RESPALDOS = "respaldos";
    private static final int RESPALDOS_A_CONSERVAR = 5;

    /** Archivo que deja la pantalla "Respaldo de datos" al importar; se aplica en el siguiente arranque. */
    public static final String ARCHIVO_IMPORTACION = "importacion-pendiente.db";

    public static void main(String[] args) {
        RegistroLog.iniciar();

        // dev al ejecutar desde Maven/IDE; producción en el programa instalado (jpackage)
        if (System.getProperty("spring.profiles.active") == null
                && System.getProperty("jpackage.app-path") == null) {
            System.setProperty("spring.profiles.active", "dev");
        }

        prepararBaseDeDatos();
        Application.launch(MisGastosApplication.class, args);
    }

    private static void prepararBaseDeDatos() {
        try {
            Path carpeta = Paths.get(System.getProperty("user.home"), "MisGastos");
            Files.createDirectories(carpeta);

            Path db = carpeta.resolve("misgastos.db");

            // Primer arranque: si no hay base (o está vacía), copiar la plantilla
            if (!Files.exists(db) || Files.size(db) == 0) {
                try (InputStream in = MisGastosLauncher.class.getResourceAsStream(PLANTILLA)) {
                    if (in != null) {
                        Files.copy(in, db, StandardCopyOption.REPLACE_EXISTING);
                        System.out.println("Base inicial creada desde la plantilla: " + db);
                    } else {
                        System.err.println("ADVERTENCIA: no se encontró la plantilla " + PLANTILLA);
                    }
                }
            }

            // Copia de seguridad ANTES de que Spring abra la base (y antes de cualquier
            // cambio de estructura). Si falla, solo se avisa: nunca impide abrir el programa.
//            respaldarBaseDeDatos(carpeta, db);
//            aplicarImportacionPendiente(carpeta, db);
            
            respaldarBaseDeDatos(carpeta, db);
            aplicarImportacionPendiente(carpeta, db);
            MigracionesBase.aplicar(db);   // <-- NUEVA

            String dbPath = db.toString().replace('\\', '/');
            System.setProperty("spring.datasource.url", "jdbc:sqlite:" + dbPath);
        } catch (Exception e) {
            throw new IllegalStateException("No se pudo preparar la base de datos", e);
        }
    }

    /**
     * Guarda una copia de la base en MisGastos\respaldos y conserva solo las últimas 5.
     * Se hace al arrancar, cuando la base todavía no está en uso, así que el archivo
     * está completo y consistente.
     */
    private static void respaldarBaseDeDatos(Path carpeta, Path db) {
        try {
            if (!Files.exists(db) || Files.size(db) == 0) {
                return; // nada que respaldar
            }

            Path carpetaRespaldos = carpeta.resolve(CARPETA_RESPALDOS);
            Files.createDirectories(carpetaRespaldos);

            String marca = LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd_HH-mm-ss"));
            Path copia = carpetaRespaldos.resolve("misgastos_" + marca + ".db");
            Files.copy(db, copia, StandardCopyOption.REPLACE_EXISTING);
            System.out.println("Respaldo creado: " + copia);

            limpiarRespaldosAntiguos(carpetaRespaldos);
        } catch (Exception e) {
            System.err.println("ADVERTENCIA: no se pudo crear el respaldo automático: " + e.getMessage());
        }
    }

    /** Borra los respaldos más viejos; el nombre lleva la fecha, así que ordenar por nombre basta. */
    private static void limpiarRespaldosAntiguos(Path carpetaRespaldos) throws java.io.IOException {
        List<Path> respaldos;
        try (Stream<Path> archivos = Files.list(carpetaRespaldos)) {
            respaldos = archivos
                    .filter(p -> {
                        String n = p.getFileName().toString();
                        return n.startsWith("misgastos_") && n.endsWith(".db");
                    })
                    .sorted()
                    .toList();
        }

        int sobrantes = respaldos.size() - RESPALDOS_A_CONSERVAR;
        for (int i = 0; i < sobrantes; i++) {
            try {
                Files.deleteIfExists(respaldos.get(i));
            } catch (Exception e) {
                System.err.println("No se pudo borrar el respaldo antiguo " + respaldos.get(i) + ": " + e.getMessage());
            }
        }
    }

    /**
     * Si el usuario importó una copia, la reemplaza ahora, antes de que Spring abra la base.
     * Ya se hizo un respaldo de la base anterior justo arriba (respaldarBaseDeDatos).
     */
    private static void aplicarImportacionPendiente(Path carpeta, Path db) {
        Path pendiente = carpeta.resolve(ARCHIVO_IMPORTACION);
        try {
            if (!Files.exists(pendiente)) {
                return;
            }
            if (Files.size(pendiente) == 0) {
                Files.deleteIfExists(pendiente);
                return;
            }
            // Archivos auxiliares de la base anterior (si existieran) no deben mezclarse con la nueva
            Files.deleteIfExists(carpeta.resolve("misgastos.db-wal"));
            Files.deleteIfExists(carpeta.resolve("misgastos.db-shm"));
            Files.deleteIfExists(carpeta.resolve("misgastos.db-journal"));

            Files.move(pendiente, db, StandardCopyOption.REPLACE_EXISTING);
            System.out.println("Copia importada y aplicada: " + db);
        } catch (Exception e) {
            System.err.println("ADVERTENCIA: no se pudo aplicar la copia importada: " + e.getMessage());
        }
    }
}
